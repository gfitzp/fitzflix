{# The one pagination strip of the site. pages is the Flask-SQLAlchemy pager. endpoint and args build each page link, and a None value in args drops out of the link. compact trims iter_pages so the strip fits a phone, and sets the breakpoint class of the nav. compact none renders one strip with no breakpoint. A page that pages a list by hand passes numbers and current instead of pages. The ellipsis of a trimmed strip is a disabled item. #} {% macro pagination(pages, endpoint, args={}, prev_url=none, next_url=none, compact=none, nav_class="", numbers=none, current=none) -%} {% set items = numbers if numbers is not none else (pages.iter_pages(left_edge=1, left_current=1, right_current=2, right_edge=1) if compact else pages.iter_pages()) %} {% set current = current if numbers is not none else pages.page %}
  • «
    • {% for page in items %} {% if page %}
  • {{ page }}
    • {% else %}
  • …
    • {% endif %} {% endfor %}
  • »
{%- endmacro %} {# A wide strip and a narrow strip at one position. Exactly one shows per breakpoint. #} {% macro pagination_pair(pages, endpoint, args={}, prev_url=none, next_url=none, nav_class="") -%} {{ pagination(pages, endpoint, args, prev_url, next_url, compact=false) }} {{ pagination(pages, endpoint, args, prev_url, next_url, compact=true, nav_class=nav_class) }} {%- endmacro %}
